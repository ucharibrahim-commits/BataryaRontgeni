package com.hardware.batterymonitor.ui

import androidx.compose.foundation.background
import androidx.compose.foundation.layout.*
import androidx.compose.foundation.shape.RoundedCornerShape
import androidx.compose.material3.*
import androidx.compose.runtime.*
import androidx.compose.ui.Modifier
import androidx.compose.ui.graphics.Color
import androidx.compose.ui.text.font.FontWeight
import androidx.compose.ui.unit.dp
import androidx.compose.ui.unit.sp

@Composable
fun BatteryDashboardScreen(viewModel: BatteryViewModel) {
    val uiState by viewModel.uiState.collectAsState()

    Column(
        modifier = Modifier
            .fillMaxSize()
            .background(Color(#0F172A)) // Endüstriyel koyu gri/mavi arka plan
            .padding(16.dp)
    ) {
        Text(
            text = "KAPALI DEVRE DONANIM ANALİZİ",
            color = Color.White,
            fontSize = 18.sp,
            fontWeight = FontWeight.Bold,
            modifier = Modifier.padding(bottom = 16.dp)
        )

        // --- RASYONEL DEĞİŞİM ALARM PANELİ ---
        if (uiState.isChangeRequired) {
            Card(
                colors = CardDefaults.cardColors(containerColor = Color(#991B1B)), // Kritik Durum Kırmızısı
                shape = RoundedCornerShape(8.dp),
                modifier = Modifier.fillMaxWidth().padding(bottom = 16.dp)
            ) {
                Column(modifier = Modifier.padding(16.dp)) {
                    Text("⚠️ FİZİKSEL BATARYA DEĞİŞİM ALARMI", color = Color.White, fontWeight = FontWeight.Bold, fontSize = 15.sp)
                    Spacer(modifier = Modifier.height(4.dp))
                    Text("Çip Analizi Sonucu: Hücrelerin kimyasal ömrü tükenmiştir.", color = Color.White, fontSize = 13.sp)
                    Text("Arıza Nedeni: ${uiState.alertReason}", color = Color(#FECACA), fontWeight = FontWeight.Bold, fontSize = 13.sp)
                }
            }
        }

        // --- ANLIK DONANIM PARAMETRE KUTULARI ---
        Row(modifier = Modifier.fillMaxWidth(), horizontalArrangement = Arrangement.spacedBy(8.dp)) {
            DataBox(label = "ANLIK AKIM ŞİDDETİ", value = "${uiState.currentInmA} mA", modifier = Modifier.weight(1f))
            DataBox(label = "HÜCRE GERİLİMİ", value = "${uiState.voltageInmV} mV", modifier = Modifier.weight(1f))
        }
        
        Spacer(modifier = Modifier.height(8.dp))

        Row(modifier = Modifier.fillMaxWidth(), horizontalArrangement = Arrangement.spacedBy(8.dp)) {
            DataBox(label = "TOPLAM ŞARJ DÖNGÜSÜ", value = "${uiState.cycleCount} Çevrim", modifier = Modifier.weight(1f))
            DataBox(label = "DONANIM SICAKLIĞI", value = "${uiState.temperature} °C", modifier = Modifier.weight(1f))
        }
    }
}

@Composable
fun DataBox(label: String, value: String, modifier: Modifier = Modifier) {
    Card(
        colors = CardDefaults.cardColors(containerColor = Color(#1E293B)),
        shape = RoundedCornerShape(6.dp),
        modifier = modifier
    ) {
        Column(modifier = Modifier.padding(14.dp)) {
            Text(text = label, color = Color(#94A3B8), fontSize = 11.sp, fontWeight = FontWeight.Medium)
            Spacer(modifier = Modifier.height(4.dp))
            Text(text = value, color = Color.White, fontSize = 17.sp, fontWeight = FontWeight.Bold)
        }
    }
}
