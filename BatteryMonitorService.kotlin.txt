package com.hardware.batterymonitor.services

import android.app.*
import android.content.Context
import android.content.Intent
import android.content.IntentFilter
import android.os.Build
import android.os.IBinder
import androidx.core.app.NotificationCompat
import com.hardware.batterymonitor.core.BatteryHardwareManager
import java.util.*

class BatteryMonitorService : Service() {

    private lateinit var hardwareManager: BatteryHardwareManager
    private var timer: Timer? = null

    override fun onCreate() {
        super.onCreate()
        hardwareManager = BatteryHardwareManager(this)
        
        // 1. Servisi ön plana çıkar ve kalıcı bildirimi durum çubuğuna sabitle
        startForegroundServiceWithNotification()
        
        // 2. Zamanlayıcı motorunu ateşle
        startDataCollectionTimer()
    }

    private fun startDataCollectionTimer() {
        timer = Timer()
        timer?.scheduleAtFixedRate(object : TimerTask() {
            override fun run() {
                // Anakart üzerindeki şarj entegresinden verileri çekiyoruz
                val currentmA = hardwareManager.getCurrentNow()
                val cycleCount = hardwareManager.getCycleCount()
                
                // [ADIM B - BAĞLANTI HATTI]: Verileri "BATTERY_HARDWARE_UPDATE" yerel frekansından fırlatıyoruz
                val updateIntent = Intent("BATTERY_HARDWARE_UPDATE").apply {
                    putExtra("current", currentmA)
                    putExtra("cycle", cycleCount)
                }
                sendBroadcast(updateIntent)
            }
        }, 0, 5000) // Her 5 saniyede bir (5000 milisaniye) ölçüm yapar
    }

    private fun startForegroundServiceWithNotification() {
        val channelId = "battery_offline_monitor_channel"
        if (Build.VERSION.SDK_INT >= Build.VERSION_CODES.O) {
            val channel = NotificationChannel(channelId, "Güvenli Pil İzleme Motoru", NotificationManager.IMPORTANCE_LOW)
            val manager = getSystemService(NotificationManager::class.java)
            manager.createNotificationChannel(channel)
        }

        val notification = NotificationCompat.Builder(this, channelId)
            .setContentTitle("Kapalı Devre Batarya Analizi Aktif")
            .setContentText("Donanım verileri %100 internetsiz ve güvenli ortamda inceleniyor.")
            .setSmallIcon(android.R.drawable.ic_lock_idle_low_battery)
            .setOngoing(true) // Kullanıcı bu bildirimi kapatamaz, servis stabil kalır
            .build()

        startForeground(1, notification)
    }

    override fun onBind(intent: Intent?): IBinder? = null

    override fun onDestroy() {
        super.onDestroy()
        timer?.cancel() // Hafıza sızıntısı olmaması için zamanlayıcıyı kapat
    }
}
